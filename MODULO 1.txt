Sub EliminarImagenes()
 Dim myobj
    Dim pictur
    Set myobj = ActiveSheet.DrawingObjects
    For Each pictur In myobj
     If Left(pictur.Name, 7) = "Picture" Then
        pictur.Select
        pictur.Delete
     End If
   Next
End Sub
